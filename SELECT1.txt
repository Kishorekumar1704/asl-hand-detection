SELECT
    'Chest' AS mod_study,
    SUM(CASE WHEN xds.organ IN ('Chest', 'Chest_Combination') THEN 1 ELSE 0 END) AS total_cases,
    SUM(CASE WHEN xds.organ IN ('Chest', 'Chest_Combination') AND (xds.rad_id IN (1504, 1505, 1506, 1627,2484,2715,2785) OR xds.primary_rad_fk IN (1504, 1505, 1506, 1627,2715,2785)) THEN 1 ELSE 0 END) AS total_reported,
    SUM(CASE WHEN xds.organ IN ('Chest', 'Chest_Combination') AND xds.primary_rad_fk IN (1504, 1505, 1506, 1627,2484,2715,2785) THEN 1 ELSE 0 END) AS total_rework,
    COALESCE(SUM(CASE WHEN xds.organ in ('Chest','Chest_Combination')  AND xds.user_name IS NOT NULL THEN 1 ELSE 0 END), 0) AS assigned_to_hil,
    COALESCE(SUM(CASE WHEN (xds.reported_by = 'AI' OR xds.is_rework = 'AI') AND xds.organ in ('Chest','Chest_Combination') AND xds.user_name IS NOT NULL THEN 1 ELSE 0 END), 0) AS reported_by_hil,
    COALESCE(SUM(CASE WHEN s.status = 'REPORTABLE' AND s.mod_study_fk = '16' AND xds.user_name IS NULL THEN 1 ELSE 0 END), 0) AS automated,
    COALESCE(SUM(CASE WHEN xds.organ IN ('Chest', 'Chest_Combination') AND xds.primary_rad_fk IN (1504, 1505, 1506, 1627,2484,2715,2785) AND xds.user_name IS NOT NULL THEN 1 ELSE 0 END), 0) AS hil_rework,
    COALESCE(SUM(CASE WHEN xds.organ IN ('Chest', 'Chest_Combination') AND xds.primary_rad_fk IN (1504, 1505, 1506, 1627,2484,2715,2785) AND xds.user_name IS NULL THEN 1 ELSE 0 END), 0) AS automated_rework,
    COALESCE(((SUM(CASE WHEN xds.organ IN ('Chest', 'Chest_Combination') AND (xds.rad_id IN (1504, 1505, 1506, 1627,2484,2715,2785) OR xds.primary_rad_fk IN (1504, 1505, 1506, 1627,2484,2715,2785)) THEN 1 ELSE 0 END) * 100) / 
    SUM(CASE WHEN xds.organ IN ('Chest', 'Chest_Combination') THEN 1 ELSE 0 END)),0) AS reported_percentage
FROM analitica.xray_daily_stats xds
LEFT JOIN study."Studies" s ON xds.study_id = s.study_fk
WHERE DATE(study_created_at) = DATE(NOW()) - 1
